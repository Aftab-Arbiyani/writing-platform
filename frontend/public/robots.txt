# Qalam — robots policy. Public content (landing, feed, discover, search, writer profiles) is
# indexable; private/authenticated + utility surfaces are disallowed. Deep app state behind auth
# redirects to login for crawlers anyway; these rules make the intent explicit.
User-agent: *
Allow: /
Disallow: /me
Disallow: /settings
Disallow: /write
Disallow: /auth/
Disallow: /401
Disallow: /403
Disallow: /404
Disallow: /offline
